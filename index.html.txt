function updateCanonical(slug, lang) {

    let canonical = document.getElementById(
        'dynamic-canonical'
    );

    if (!canonical) {

        canonical = document.createElement('link');

        canonical.id = 'dynamic-canonical';

        canonical.rel = 'canonical';

        document.head.appendChild(canonical);

    canonical.href =
        window.location.origin +
        window.location.pathname +
        '?v=' + slug +
        '&lang=' + lang;

function seoUrl(text) {
    return text
        .toLowerCase()
        .replace(/[^\w\s-]/g, '')
        .replace(/\s+/g, '-')
        .replace(/--+/g, '-')
        .substring(0, 70)
        .replace(/-$/, '');
}

/* =========================================
   LANGUAGE
========================================= */

function changeLang(lang) {
    currentLang = lang;

    const currentSlug = getCurrentSlug(); // slug saat ini
    let targetSlug = null;

    if (currentSlug) {
        // pilih database target dan source
        const dbTarget = lang === 'id' ? videoDatabaseID : videoDatabaseEN;
        const dbSource = lang === 'id' ? videoDatabaseEN : videoDatabaseID;

        // cari video asal berdasarkan slug
        const videoSource = dbSource.find(v => v.slug === currentSlug);

        if (videoSource) {
            // cari versi target berdasarkan youtubeId
            const videoTarget = dbTarget.find(v => v.youtubeId === videoSource.youtubeId);
            if (videoTarget) {
                targetSlug = videoTarget.slug;
            } else {
                // fallback: pakai slug asal supaya video tetap muncul
                targetSlug = currentSlug;
                console.warn(`⚠ Video target bahasa ${lang} tidak ditemukan, fallback ke slug asal.`);
            }
        } else {
            // fallback: pakai slug asal
            targetSlug = currentSlug;
            console.warn(`⚠ Video asal tidak ditemukan, fallback ke slug.`);
        }
    }

    // bangun URL
    let url = window.location.pathname + '?lang=' + lang;
    if (targetSlug) url += '&v=' + targetSlug;

    history.pushState({}, '', url);

    // panggil router dengan targetSlug
    router(targetSlug);
}

// Fungsi search
function doSearch() {
    const val = searchInput.value.toLowerCase();

    // Pilih database sesuai bahasa
    const db = (typeof currentLang !== 'undefined' && currentLang === 'id')
        ? videoDatabaseID
        : videoDatabaseEN;

    if (!db || !Array.isArray(db)) {
        console.warn("Database video belum tersedia!");
        return;
    }

    // Filter video
    const filtered = db.filter(v => v.title.toLowerCase().includes(val));

    // Render hasil
    renderGrid(document.getElementById('app'), filtered);
}

/* =========================================
   PLAYER
========================================= */

function handlePlay(id) {

    playCounter++;

    if (
        PLAYER_CONFIG.enablePlayButtonDirectAds &&
        playCounter === 1 &&
        AD_CONFIG.allAds &&
        AD_CONFIG.useDirect
    ) {

        runAd('direct');

        document.getElementById(
            'play-msg'
        ).innerText = '';

        return;
    }


    document.getElementById(
        'player-box'
    ).innerHTML =

    '<iframe src="https://www.youtube.com/embed/' +
    id +
    '?autoplay=1&rel=0&modestbranding=1&fs=0&controls=1&playsinline=1" allow="autoplay; encrypted-media; fullscreen" allowfullscreen webkitallowfullscreen mozallowfullscreen style="width:100%; height:100%; border:none;"></iframe>' +

    '<div class="video-mask mask-top"></div>' +

    '<div class="video-mask mask-bottom"></div>' +

    '<div id="fs-btn" class="btn-fs-custom" onclick="toggleFullScreen()"><i class="fas fa-expand"></i></div>';
}


function toggleFullScreen() {

    const elem = document.getElementById(
        'player-box'
    );

    const icon = document.querySelector(
        '#fs-btn i'
    );

    if (
        !document.fullscreenElement &&
        !document.webkitFullscreenElement
    ) {

        if (elem.requestFullscreen) {
            elem.requestFullscreen();
        }

        else if (elem.webkitRequestFullscreen) {
            elem.webkitRequestFullscreen();
        }

        icon.className = 'fas fa-compress';

    } else {

        if (document.exitFullscreen) {
            document.exitFullscreen();
        }

        else if (document.webkitExitFullscreen) {
            document.webkitExitFullscreen();
        }

        icon.className = 'fas fa-expand';
    }
}

function setOGTag(property, content) {

    let tag = document.querySelector(
        'meta[property="' + property + '"]'
    );

    if (!tag) {

        tag = document.createElement('meta');

        tag.setAttribute(
            'property',
            property
        );

        document.head.appendChild(tag);
    }

    tag.setAttribute(
        'content',
        content
    );
}



function updateMetaTags(video) {

    if (!descTag) {

        descTag.setAttribute(
            'name',
            'description'
        );

        document.head.appendChild(descTag);
    }

    descTag.setAttribute(
        'content',
        video.summary || video.title
    );
}

   ROUTER
========================================= */

async function router() {

    const app = document.getElementById('app');

    let slug = getCurrentSlug();

    let lang = getCurrentLangParam();


    if (!lang) {

        if (SEO_CONFIG.autoGeoLanguage) {

            lang = await detectGeoLanguage();

        } else {

            lang = 'id';
        }
    }


    currentLang = lang;


    document.getElementById('btn-id').className =
        lang === 'id'
            ? 'btn-lang active'
            : 'btn-lang';


    document.getElementById('btn-en').className =
        lang === 'en'
            ? 'btn-lang active'
            : 'btn-lang';


    const db =
        lang === 'id'
            ? videoDatabaseID
            : videoDatabaseEN;


    if (!slug || slug === 'home') {

        renderGrid(
            app,
            [...db].sort(() => 0.5 - Math.random())
        );

        updateCanonical('home', lang);

    } else {

        const video = db.find(
    v =>
        v.slug === slug ||
        seoUrl(v.title) === slug
);

        if (video) {

    console.log(video);

    renderVideo(app, video);

    updateCanonical(video.slug, lang);

    updateMetaTags(video);

} else {

    document.title = "Trend4GenZ - Streaming";

    renderGrid(app, db);

    updateCanonical('home', lang);
}
    }


    window.scrollTo(0,0);
}
